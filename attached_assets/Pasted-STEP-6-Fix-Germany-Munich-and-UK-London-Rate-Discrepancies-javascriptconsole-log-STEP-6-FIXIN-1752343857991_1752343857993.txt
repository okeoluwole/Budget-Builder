function fixHighVarianceLocations() {
    console.log("Addressing Germany/Munich and UK/London rate discrepancies...");
    
    // Target the two locations with >20% variance
    const problemLocations = [
        { country: "Germany", city: "Munich", currentVariance: "21.2%" },
        { country: "United Kingdom", city: "London", currentVariance: "21.3%" }
    ];
    
    problemLocations.forEach(location => {
        console.log(`\nDebugging ${location.country}/${location.city}:`);
        
        // 1. Verify current rates being used
        const currentRates = getCurrentTTRates(location.city, location.country);
        console.log(`Current rates for ${location.city}:`, currentRates);
        
        // 2. Extract rates directly from PDF source for this location
        const pdfRates = extractPDFRatesForLocation(location.city, location.country);
        console.log(`PDF source rates for ${location.city}:`, pdfRates);
        
        // 3. Compare component by component
        ["Construction", "Loose Furniture", "Audio-Visual"].forEach(component => {
            const currentRate = currentRates[component];
            const pdfRate = pdfRates[component];
            const variance = Math.abs((currentRate - pdfRate) / pdfRate * 100);
            
            console.log(`${component}: Current $${currentRate} vs PDF $${pdfRate} (${variance.toFixed(1)}% variance)`);
            
            if (variance > 15) {
                console.log(`❌ ${component} needs rate adjustment for ${location.city}`);
                // Update the rate to match PDF source
                updateTTComponentRate(location.city, location.country, component, pdfRate);
            }
        });
        
        // 4. Recalculate and verify
        const updatedResult = calculateTTLocationTotal(location.city, location.country);
        console.log(`Updated calculation for ${location.city}:`, updatedResult);
    });
}

// Execute rate fixes
fixHighVarianceLocations();
