// JLL cost calculation function
function calculateJllRegionalCost(region, layoutIndex, qualityIndex, area) {
  const regionData = jllData.regions[region];
  
  if (!regionData) {
    throw new Error(`No data available for region: ${region}`);
  }
  
  // For regional data, we use the layout type to get the base cost
  const baseCost = regionData.costs[layoutIndex];
  
  // Apply quality multiplier based on global averages
  const qualityMultiplier = jllData.global.costs[layoutIndex][qualityIndex] / 
                           jllData.global.costs[layoutIndex][1]; // Medium is the baseline
  
  const totalCost = baseCost * qualityMultiplier * area;
  
  // Calculate cost breakdown
  const breakdown = {
    buildersWorks: totalCost * regionData.breakdown.buildersWorks,
    meServices: totalCost * regionData.breakdown.meServices,
    ffe: totalCost * regionData.breakdown.ffe,
    securityItAv: totalCost * regionData.breakdown.securityItAv,
    professionalServices: totalCost * regionData.breakdown.professionalServices,
    reinstatement: regionData.reinstatement * area
  };
  
  return {
    totalCost,
    breakdown,
    costPerSqm: totalCost / area,
    reinstatementCost: regionData.reinstatement * area
  };
}
