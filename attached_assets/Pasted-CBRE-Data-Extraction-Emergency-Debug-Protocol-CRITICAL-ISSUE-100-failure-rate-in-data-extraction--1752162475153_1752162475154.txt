CBRE Data Extraction Emergency Debug Protocol
CRITICAL ISSUE: 100% failure rate in data extraction. The system is providing false confirmations while extracting incorrect data.
Objective: Create a systematic, step-by-step debugging approach to fix CBRE data extraction from the ground up.
Specific Instructions for REPLIT:
Phase 1: Basic File Access Verification
Step 1: Verify File Existence and Basic Reading
1. First, confirm the exact files are accessible:

   async function verifyFileAccess() {
       console.log("=== STEP 1: BASIC FILE ACCESS VERIFICATION ===");
       
       const expectedFiles = [
           'CBRE Fit Out Guide Cost Data_Fitout Only.xlsx',
           'CBRE Fit Out Guide Cost Data_Furniture Only.xlsx', 
           'CBRE Fit Out Guide Cost Data_29.01.2024_Tech Only.xlsx'
       ];
       
       const fileAccessResults = {};
       
       for (const fileName of expectedFiles) {
           try {
               console.log(`Testing access to: ${fileName}`);
               const fileData = await window.fs.readFile(fileName);
               
               fileAccessResults[fileName] = {
                   accessible: true,
                   size: fileData.length,
                   type: typeof fileData
               };
               
               console.log(`✅ ${fileName}: Accessible (${fileData.length} bytes)`);
               
           } catch (error) {
               fileAccessResults[fileName] = {
                   accessible: false,
                   error: error.message
               };
               
               console.error(`❌ ${fileName}: ${error.message}`);
           }
       }
       
       console.log("\nFile Access Summary:", fileAccessResults);
       
       // Stop here if any files are inaccessible
       const allAccessible = Object.values(fileAccessResults).every(result => result.accessible);
       if (!allAccessible) {
           throw new Error("Some CBRE files are not accessible. Fix file access before proceeding.");
       }
       
       return fileAccessResults;
   }
   
   // Execute this step first
   const fileAccess = await verifyFileAccess();

   async function analyzeWorkbookStructures() {
       console.log("=== STEP 2: WORKBOOK STRUCTURE ANALYSIS ===");
       
       const structures = {};
       
       const files = [
           'CBRE Fit Out Guide Cost Data_Fitout Only.xlsx',
           'CBRE Fit Out Guide Cost Data_Furniture Only.xlsx', 
           'CBRE Fit Out Guide Cost Data_29.01.2024_Tech Only.xlsx'
       ];
       
       for (const fileName of files) {
           console.log(`\n--- Analyzing ${fileName} ---`);
           
           try {
               const fileData = await window.fs.readFile(fileName);
               const workbook = XLSX.read(fileData, {
                   cellStyles: true,
                   cellFormulas: true,
                   cellDates: true,
                   cellNF: true,
                   sheetStubs: true
               });
               
               console.log(`Sheets found: ${workbook.SheetNames.join(', ')}`);
               
               structures[fileName] = {
                   sheetNames: workbook.SheetNames,
                   sheetCount: workbook.SheetNames.length
               };
               
               // For each sheet, just get basic info
               workbook.SheetNames.forEach(sheetName => {
                   const sheet = workbook.Sheets[sheetName];
                   const range = sheet['!ref'];
                   console.log(`  Sheet "${sheetName}": Range ${range || 'N/A'}`);
               });
               
           } catch (error) {
               console.error(`Failed to analyze ${fileName}:`, error);
               structures[fileName] = { error: error.message };
           }
       }
       
       return structures;
   }
   
   // Execute this step
   const structures = await analyzeWorkbookStructures();
