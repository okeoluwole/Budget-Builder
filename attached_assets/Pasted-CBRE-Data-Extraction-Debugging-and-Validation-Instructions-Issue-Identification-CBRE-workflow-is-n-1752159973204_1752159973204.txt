   function extractAccurateCBREData() {
       console.log("=== IMPLEMENTING ACCURATE CBRE DATA EXTRACTION ===");
       
       // Technology data extraction
       function extractTechnologyRates(location, layoutType, specLevel) {
           console.log(`Extracting tech rates: ${location}-${layoutType}-${specLevel}`);
           
           try {
               // Use Technology-specific file
               const techWorkbook = getTechnologyWorkbook();
               let targetSheet;
               
               // Determine correct sheet based on layout type
               switch(layoutType.toLowerCase()) {
                   case 'activity':
                       targetSheet = techWorkbook.Sheets['Table 1_Activity BYOB Sqm'];
                       break;
                   case 'team':
                       targetSheet = techWorkbook.Sheets['Table 2_Team BYOB Sqm'];
                       break;
                   case 'event':
                       targetSheet = techWorkbook.Sheets['Table 3_Event BYOB Sqm'];
                       break;
                   default:
                       targetSheet = techWorkbook.Sheets['TECH']; // Combined sheet
               }
               
               const sheetData = XLSX.utils.sheet_to_json(targetSheet, {header: 1});
               
               // Find London row
               const londonRow = sheetData.find(row => 
                   row[1] && row[1].toString().toLowerCase().includes('kingdom') &&
                   row[2] && row[2].toString().toLowerCase().includes('london')
               );
               
               if (!londonRow) {
                   throw new Error(`London not found in technology data for ${layoutType}`);
               }
               
               // Extract rate based on specification level
               let rateIndex;
               switch(specLevel.toLowerCase()) {
                   case 'low':
                   case 'bronze':
                       rateIndex = 4; // Column E
                       break;
                   case 'medium':
                   case 'silver':
                       rateIndex = 5; // Column F
                       break;
                   case 'high':
                   case 'gold':
                       rateIndex = 6; // Column G
                       break;
                   default:
                       throw new Error(`Invalid specification level: ${specLevel}`);
               }
               
               const rate = londonRow[rateIndex];
               console.log(`Technology rate extracted: $${rate}/sqm for ${specLevel} ${layoutType}`);
               return rate;
               
           } catch (error) {
               console.error("Technology rate extraction failed:", error);
               throw error;
           }
       }
       
       // Fitout data extraction
       function extractFitoutRates(location, layoutType, specLevel) {
           console.log(`Extracting fitout rates: ${location}-${layoutType}-${specLevel}`);
           
           try {
               const fitoutWorkbook = getFitoutWorkbook();
               const targetSheet = fitoutWorkbook.Sheets['FITOUT'] || fitoutWorkbook.Sheets[0];
               const sheetData = XLSX.utils.sheet_to_json(targetSheet, {header: 1});
               
               // Find London row
               const londonRow = sheetData.find(row => 
                   row[1] && row[1].toString().toLowerCase().includes('kingdom') &&
                   row[2] && row[2].toString().toLowerCase().includes('london')
               );
               
               if (!londonRow) {
                   throw new Error("London not found in fitout data");
               }
               
               // Determine column index based on layout and specification
               let rateIndex;
               if (layoutType.toLowerCase() === 'activity') {
                   switch(specLevel.toLowerCase()) {
                       case 'low':
                       case 'bronze':
                           rateIndex = 4;
                           break;
                       case 'medium':
                       case 'silver':
                           rateIndex = 5;
                           break;
                       case 'high':
                       case 'gold':
                           rateIndex = 6;
                           break;
                   }
               } else if (layoutType.toLowerCase() === 'team') {
                   switch(specLevel.toLowerCase()) {
                       case 'low':
                       case 'bronze':
                           rateIndex = 7;
                           break;
                       case 'medium':
                       case 'silver':
                           rateIndex = 8;
                           break;
                       case 'high':
                       case 'gold':
                           rateIndex = 9;
                           break;
                   }
               } else if (layoutType.toLowerCase() === 'event') {
                   rateIndex = 10; // Event has single rate
               }
               
               const rate = londonRow[rateIndex];
               console.log(`Fitout rate extracted: $${rate}/sqm for ${specLevel} ${layoutType}`);
               return rate;
               
           } catch (error) {
               console.error("Fitout rate extraction failed:", error);
               throw error;
           }
       }
       
       return { extractTechnologyRates, extractFitoutRates };
   }

   function implementCorrectedRateLookup() {

       function getCBRERateAccurate(location, component, specification, layoutType = 'Activity') {
           console.log(`Getting accurate CBRE rate: ${location}-${component}-${specification}-${layoutType}`);
           
           try {
               let rate;
               
               switch(component.toLowerCase()) {
                   case 'fitout':
                   case 'fitout construction':

                       rate = extractFitoutRates(location, layoutType, specification);
                       break;
                       
                   case 'furniture':
                   case 'loose furniture':
                       rate = extractFurnitureRates(location, layoutType, specification);
                       break;
                       
                   case 'technology':
                   case 'tech':
                   case 'av':
                   case 'audio-visual':
                       rate = extractTechnologyRates(location, layoutType, specification);
                       break;
                       
                   case 'professional fees':
                   case 'prof fees':
                       // Professional fees are percentage-based, not per sqm
                       rate = getProfessionalFeesPercentage(location);
                       break;
                       
                   default:
                       throw new Error(`Unknown component: ${component}`);
               }
               
               if (!rate || rate <= 0) {
                   throw new Error(`Invalid rate retrieved: ${rate} for ${component}`);
               }
               
               console.log(`Accurate rate retrieved: $${rate}/sqm for ${component}`);
               return rate;
               
           } catch (error) {
               console.error(`Rate lookup failed for ${component}:`, error);
               throw error;
           }
       }
       
       return getCBRERateAccurate;
   }

   function implementAccurateProfessionalFees() {
       console.log("=== IMPLEMENTING ACCURATE PROFESSIONAL FEES ===");
       
       function getProfessionalFeesPercentage(location) {
           // Professional fees percentages by location (from CBRE data)
           const profFeesPercentages = {
               'London': 4.9,
               'New York': 5.2,
               'Singapore': 4.8,
               'Tokyo': 5.1,
               'Sydney': 4.7,
               'Paris': 5.0,
               'Dubai': 4.6
           };
           
           const percentage = profFeesPercentages[location] || 5.0; // Default 5%
           console.log(`Professional fees percentage for ${location}: ${percentage}%`);
           return percentage;
       }
       
       function calculateProfessionalFees(fitoutCost, furnitureCost, technologyCost, location) {
           const subtotal = fitoutCost + furnitureCost + technologyCost;
           const percentage = getProfessionalFeesPercentage(location);
           const professionalFees = subtotal * (percentage / 100);
           
           console.log(`Professional fees calculation:`);
           console.log(`  Subtotal: $${subtotal}`);
           console.log(`  Percentage: ${percentage}%`);
           console.log(`  Professional Fees: $${professionalFees}`);
           
           return {
               amount: professionalFees,
               percentage: percentage,
               ratePerSqm: professionalFees // Will be divided by area in display
           };
       }
       
       return { getProfessionalFeesPercentage, calculateProfessionalFees };
   }

    function applyCorrectionsToLiveSystem() {
        console.log("=== APPLYING CORRECTIONS TO LIVE SYSTEM ===");

        window.getCBRERate = getCBRERateAccurate;
        window.calculateProfessionalFees = calculateProfessionalFees;
        window.extractTechnologyRates = extractTechnologyRates;
        window.extractFitoutRates = extractFitoutRates;
        
        // Update calculation engine
        function updateCalculationEngine() {
            const originalCalculateCompletePackage = window.calculateCompletePackage;
            
            window.calculateCompletePackage = function(inputs) {
                console.log("Using corrected calculation engine");
                
                try {
                    const { location, area, fitoutSpec, furnitureSpec, technologySpec, layoutType = 'Activity' } = inputs;
                    
                    // Use corrected rate extraction
                    const fitoutRate = getCBRERateAccurate(location, 'Fitout', fitoutSpec, layoutType);
                    const furnitureRate = getCBRERateAccurate(location, 'Furniture', furnitureSpec, layoutType);  
                    const technologyRate = getCBRERateAccurate(location, 'Technology', technologySpec, layoutType);
                    
                    const fitoutCost = fitoutRate * area;
                    const furnitureCost = furnitureRate * area;
                    const technologyCost = technologyRate * area;
                    
                    const profFees = calculateProfessionalFees(fitoutCost, furnitureCost, technologyCost, location);
                    const professionalFeesCost = profFees.amount;
                    
                    const totalCost = fitoutCost + furnitureCost + technologyCost + professionalFeesCost;
                    
                    return {
                        totalCost: totalCost,
                        components: [
                            { name: 'Fitout Construction', cost: fitoutCost, rate: fitoutRate, specification: fitoutSpec },
                            { name: 'Furniture', cost: furnitureCost, rate: furnitureRate, specification: furnitureSpec },
                            { name: 'Technology', cost: technologyCost, rate: technologyRate, specification: technologySpec },
                            { name: 'Professional Fees', cost: professionalFeesCost, rate: professionalFeesCost/area, percentage: profFees.percentage }
                        ],
                        breakdown: {
                            subtotal: fitoutCost + furnitureCost + technologyCost,
                            professionalFees: professionalFeesCost,
                            totalCost: totalCost,
                            ratePerSqm: totalCost / area
                        }
                    };
                    
                } catch (error) {
                    console.error("Corrected calculation failed:", error);
                    // Fallback to original if correction fails
                    return originalCalculateCompletePackage(inputs);
                }
            };
        }
        
        updateCalculationEngine();
        
        console.log("✅ Corrections applied to live system");
    }
