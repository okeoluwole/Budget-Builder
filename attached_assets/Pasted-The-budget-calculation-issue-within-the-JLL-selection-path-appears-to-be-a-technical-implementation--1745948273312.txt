javascript// Standardize area unit conversions
function convertToSquareMeters(area, fromUnit) {
  if (!area || area <= 0) {
    throw new Error("Invalid area value");
  }
  
  switch (fromUnit?.toLowerCase()) {
    case 'sqm':
    case 'square meters':
    case 'm2':
      return area; // Already in square meters
    case 'sqft':
    case 'square feet':
    case 'ft2':
      return area * 0.092903; // Convert sq ft to sq m
    default:
      console.warn(`Unknown area unit: ${fromUnit}, assuming square meters`);
      return area;
  }
}
