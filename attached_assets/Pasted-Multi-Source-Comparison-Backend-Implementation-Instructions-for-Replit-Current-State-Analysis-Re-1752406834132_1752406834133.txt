```javascript
// Create comprehensive mapping between guide nomenclatures:
const guideMapping = {

  },
  componentMapping: {
    'cbre': ['Construction', 'Furniture', 'Technology', 'Professional Fees'],
    'tt': ['Base Build', 'FF&E', 'IT/AV', 'Design Fees'],
    'jll': ['Fit-out', 'Furniture', 'Technology', 'Consultancy']
  }

async function calculateGuideWithMapping(guideId, projectData) {
  // Check country data availability first
  const hasCountryData = await checkCountryDataAvailability(guideId, projectData.country);
  
  if (!hasCountryData) {
    return {
      result: null,
      assumptions: [`No data available for ${projectData.country} in ${getGuideName(guideId)}`],
      coverage: 'unavailable'
    };
  }
  
  // Map specification levels
  const mappedSpec = mapSpecificationLevel(projectData.specLevel, guideId);
  const mappedProject = {
    ...projectData,
    specLevel: mappedSpec
  };
  
  // Calculate with mapped parameters
  const result = await calculateGuide(guideId, mappedProject);
  
  // Document assumptions made
  const assumptions = [
    `Specification mapping: ${projectData.specLevel} → ${mappedSpec}`,
    `Component mapping: ${documentComponentMapping(guideId)}`,
    `Currency: ${result.currency}`,
    `Data vintage: ${getDataVintage(guideId)}`
  ];
  
  return {
    result,
    assumptions,
    coverage: 'available'
  };
}
```

**6. Country-Level Data Availability Check**
```javascript
// Implement data availability checking:
const countryDataMatrix = {
  'cbre': ['UK', 'USA', 'Singapore', 'Australia', 'Germany', 'France', /* ... */],
  'tt': ['UK', 'USA', 'Singapore', 'Australia', 'UAE', /* ... */],
  'jll': ['UK', 'USA', 'Singapore', 'Australia', 'Japan', /* ... */]
};

function checkCountryDataAvailability(guideId, country) {
  return countryDataMatrix[guideId]?.includes(country) || false;
}

// Display data availability warnings:
function generateDataAvailabilityWarnings(comparisonGuides, country) {
  const warnings = [];
  
  comparisonGuides.forEach(guideId => {
    if (!checkCountryDataAvailability(guideId, country)) {
      warnings.push(`${getGuideName(guideId)}: No data available for ${country}`);
    }
  });
  
  return warnings;
}
```

### Phase 4: Professional Variance Analysis

**7. Add Currency-Aware Comparison**
```javascript
// Handle currency differences in comparison:
function calculateVariance(primaryCost, primaryCurrency, comparisonCost, comparisonCurrency) {
  if (!comparisonCost) return 'No data';
  
  // Convert to common currency if different
  let normalizedComparison = comparisonCost;
  if (primaryCurrency !== comparisonCurrency) {
    normalizedComparison = convertCurrency(comparisonCost, comparisonCurrency, primaryCurrency);
  }
  
  const variance = ((normalizedComparison - primaryCost) / primaryCost) * 100;
  return {
    variance: variance.toFixed(1) + '%',
    direction: variance > 0 ? 'higher' : 'lower',
    converted: primaryCurrency !== comparisonCurrency,
    note: primaryCurrency !== comparisonCurrency ? 
      `Converted from ${comparisonCurrency} to ${primaryCurrency}` : null
  };
}
```
