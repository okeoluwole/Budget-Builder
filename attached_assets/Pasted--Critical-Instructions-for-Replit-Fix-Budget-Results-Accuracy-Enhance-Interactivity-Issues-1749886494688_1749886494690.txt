## 3. Interactive JavaScript Functionality

### **Add Interactive Behaviors:**

```javascript
// Interactive hover effects for cost rows
document.querySelectorAll('.cost-row.interactive').forEach(row => {
  row.addEventListener('mouseenter', (e) => {
    const category = e.target.dataset.category;
    highlightCategory(category);
    showCategoryTooltip(category, e.target);
  });
  
  row.addEventListener('mouseleave', (e) => {
    clearHighlights();
    hideTooltip();
  });
});

// View switching functionality
document.querySelectorAll('.view-btn').forEach(btn => {
  btn.addEventListener('click', (e) => {
    const view = e.target.dataset.view;
    switchView(view);
  });
});

// Chart initialization
function initializeBudgetChart() {
  const ctx = document.getElementById('budget-pie-chart').getContext('2d');
  new Chart(ctx, {
    type: 'doughnut',
    data: {
      labels: ['Construction', 'Furniture', 'Technology', 'Professional'],
      datasets: [{
        data: [72.9, 12.7, 9.5, 4.9],
        backgroundColor: ['#3B82F6', '#10B981', '#F59E0B', '#8B5CF6'],
        borderWidth: 0,
        cutout: '60%'
      }]
    },
    options: {
      responsive: true,
      plugins: {
        legend: {
          display: false
        }
      }
    }
  });
}
```
