```
   function validateCBREDataIntegrity() {
       const requiredLocations = ["London", "New York", "Singapore"];
       const requiredComponents = ["Fitout", "Furniture", "Technology"];
       const requiredSpecLevels = ["Low", "Medium", "High"];
       
       let validationResults = {
           locationsValid: true,
           componentsValid: true,
           ratesValid: true,
           errors: []
       };
       
       // Test each combination
       requiredLocations.forEach(location => {
           requiredComponents.forEach(component => {
               requiredSpecLevels.forEach(specLevel => {
                   try {
                       const rate = getCBRERate(location, component, specLevel);
                       if (!rate || rate <= 0) {
                           validationResults.errors.push(`Missing rate: ${location}-${component}-${specLevel}`);
                           validationResults.ratesValid = false;
                       }
                   } catch (error) {
                       validationResults.errors.push(`Error accessing: ${location}-${component}-${specLevel}: ${error.message}`);
                   }
               });
           });
       });
       
       return validationResults;
   }
```
