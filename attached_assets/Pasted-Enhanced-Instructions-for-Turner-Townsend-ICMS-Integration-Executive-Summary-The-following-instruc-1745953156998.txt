Create a comprehensive calculation function that properly handles T&T's data structure:
javascript// Main calculation function for T&T data
function calculateTurnerTownsendBudget(inputs) {
  console.log("Starting T&T budget calculation with inputs:", inputs);
  
  try {
    // Destructure inputs with defaults
    const { 
      city, 
      buildingType, 
      area = 0, 
      areaUnit = 'sqm', 
      adjustForInflation = false, 
      targetYear = new Date().getFullYear(), 
      includePrelims = true, 
      includeMargin = true 
    } = inputs;
    
    // Validate required inputs
    if (!city) throw new Error("City is required");
    if (!buildingType) throw new Error("Building type is required");
    if (!area || isNaN(parseFloat(area)) || parseFloat(area) <= 0) {
      throw new Error("Area must be a positive number");
    }
    
    // Get city data
    const cityData = window.TurnerTownsendData.getCityData(city);
    if (!cityData) {
      throw new Error(`No data available for city: ${city}`);
    }
    
    console.log("City data found:", city);
    
    // Get building cost data
    const buildingCosts = cityData.buildingCosts[buildingType];
    if (!buildingCosts) {
      throw new Error(`No data available for building type: ${buildingType} in ${city}`);
    }
    
    console.log("Building cost data found:", buildingCosts);
    
    // Determine cost per unit based on area unit
    const costPerUnit = areaUnit.toLowerCase() === 'sqm' ? 
      buildingCosts.costPerSqM : buildingCosts.costPerSqFt;
    
    console.log(`Base cost per ${areaUnit}: $${costPerUnit}`);
    
    // Parse area to ensure it's a number
    const numericArea = parseFloat(area);
    
    // Calculate base cost
    let baseCost = costPerUnit * numericArea;
    console.log(`Base cost calculation: ${costPerUnit} × ${numericArea} = ${baseCost}`);
    
    // Apply inflation if requested
    let appliedInflation = 0;
    if (adjustForInflation && targetYear) {
      const currentYear = new Date().getFullYear();
      const yearDiff = targetYear - currentYear;
      
      if (yearDiff > 0) {
        console.log(`Applying inflation for ${yearDiff} years ahead`);
        
        // Compound inflation calculation
        let inflationMultiplier = 1.0;
        
        // Apply each year's inflation rate
        for (let i = 0; i < yearDiff; i++) {
          const yearToApply = currentYear + i;
          let yearRate;
          
          if (yearToApply === currentYear) {
            yearRate = cityData.inflation['2024'] || 0;
          } else if (yearToApply === currentYear + 1) {
            yearRate = cityData.inflation['2025'] || 0;
          } else {
            // For years beyond available data, use the latest available
            yearRate = cityData.inflation['2025'] || 0;
          }
          
          inflationMultiplier *= (1 + yearRate);
          console.log(`Year ${yearToApply} inflation: ${(yearRate * 100).toFixed(1)}%, multiplier: ${inflationMultiplier.toFixed(4)}`);
        }
        
        appliedInflation = baseCost * (inflationMultiplier - 1);
        baseCost *= inflationMultiplier;
        
        console.log(`After inflation adjustment: ${baseCost}`);
      }
    }
    
    // Add preliminaries if requested
    let prelimsCost = 0;
    if (includePrelims) {
      prelimsCost = baseCost * cityData.preliminaries;
      console.log(`Preliminaries (${(cityData.preliminaries * 100).toFixed(1)}%): ${prelimsCost}`);
    }
    
    // Add margin if requested
    let marginCost = 0;
    if (includeMargin) {
      marginCost = (baseCost + prelimsCost) * cityData.contractorMargin;
      console.log(`Contractor's margin (${(cityData.contractorMargin * 100).toFixed(1)}%): ${marginCost}`);
    }
    
    // Calculate total
    const totalCost = baseCost + prelimsCost + marginCost;
    console.log(`Total cost: ${totalCost}`);
    
    // Format amounts for display
    const formatAmount = (amount) => Math.round(amount);
    
    // Create detailed breakdown
    const breakdown = {
      baseBuildingCost: formatAmount(baseCost - appliedInflation),
      inflationAdjustment: formatAmount(appliedInflation),
      adjustedBaseCost: formatAmount(baseCost),
      preliminaries: formatAmount(prelimsCost),
      margin: formatAmount(marginCost),
      total: formatAmount(totalCost),
      costPerUnit: formatAmount(totalCost / numericArea),
      marketCondition: cityData.tenderingCondition,
      marketOutlook: cityData.marketOutlook,
      inflationForecast: {
        current: (cityData.inflation['2024'] * 100).toFixed(1), // Convert to percentage
        nextYear: (cityData.inflation['2025'] * 100).toFixed(1)
      },
      // Additional context data
      locationIndex: cityData.locationIndex,
      laborRates: {
        skilledTradesman: cityData.laborCosts.plumber_electrician.rate,
        siteForeman: cityData.laborCosts.site_foreman.rate
      }
    };
    
    console.log("T&T budget calculation completed successfully");
    
    return {
      success: true,
      data: breakdown
    };
    
  } catch (error) {
    console.error("T&T calculation error:", error);
    return {
      success: false,
      error: error.message || "An error occurred during calculation",
      details: error.stack
    };
  }
}
