   async function extractComponentRate(city, component, specification, layoutType, dataSource) {
       console.log(`Extracting rate: ${city}-${component}-${specification}-${layoutType}`);
       
       try {
           // Load the appropriate workbook
           const fileData = await window.fs.readFile(dataSource.file);
           const workbook = XLSX.read(fileData, {
               cellStyles: true,
               cellFormulas: true,
               cellDates: true,
               cellNF: true,
               sheetStubs: true
           });
           
           // Get the appropriate sheet
           let targetSheet;
           if (component === 'Technology') {
               // Technology has layout-specific sheets
               const sheetMap = {
                   'Activity': 'Table 1_Activity BYOB Sqm',
                   'Team': 'Table 2_Team BYOB Sqm', 
                   'Event': 'Table 3_Event BYOB Sqm'
               };
               targetSheet = workbook.Sheets[sheetMap[layoutType]] || workbook.Sheets['TECH'];
           } else {
               // Use the sheet specified in dataSource or first available
               targetSheet = workbook.Sheets[dataSource.sheet] || 
                            workbook.Sheets[Object.keys(workbook.Sheets)[0]];
           }
           
           if (!targetSheet) {
               throw new Error(`Sheet not found for ${component}`);
           }
           
           const sheetData = XLSX.utils.sheet_to_json(targetSheet, {header: 1});
           
           // Find the city's data row
           const cityRow = sheetData.find(row => {
               return row[2] && row[2].toString().toLowerCase().includes(city.toLowerCase());
           });
           
           if (!cityRow) {
               throw new Error(`City ${city} not found in ${component} data`);
           }
           
           // Determine rate column based on specification
           let rateColumnIndex;
           const specMapping = {
               'Low': 4,    // Column E
               'Medium': 5, // Column F  
               'High': 6    // Column G
           };
           
           rateColumnIndex = specMapping[specification];
           
           // For layout-specific data (like Fitout), adjust column index
           if (component === 'Fitout' && layoutType !== 'Activity') {
               if (layoutType === 'Team') {
                   rateColumnIndex += 3; // Shift to Team columns
               } else if (layoutType === 'Event') {
                   rateColumnIndex = 10; // Event column
               }
           }
           
           const rate = cityRow[rateColumnIndex];
           
           if (!rate || rate <= 0) {
               throw new Error(`Invalid rate: ${rate} for ${city}-${component}-${specification}`);
           }
           
           console.log(`Rate extracted: $${rate}/sqm`);
           return rate;
           
       } catch (error) {
           console.error(`Rate extraction failed:`, error);
           throw error;
       }
   }

   function validateGlobalRateConsistency(extractionResults) {
       console.log("=== VALIDATING GLOBAL RATE CONSISTENCY ===");
       
       const consistencyAnalysis = {
           rateRanges: {},
           outliers: [],
           missingRates: [],
           averagesByRegion: {}
       };
       
       // Analyze rate ranges by component and specification
       const allRates = {
           Technology: { Low: [], Medium: [], High: [] },
           Fitout: { Low: [], Medium: [], High: [] },
           Furniture: { Low: [], Medium: [], High: [] }
       };
       
       extractionResults.successful.forEach(location => {
           const region = location.region;
           
           if (!consistencyAnalysis.averagesByRegion[region]) {
               consistencyAnalysis.averagesByRegion[region] = {
                   Technology: { Low: [], Medium: [], High: [] },
                   Fitout: { Low: [], Medium: [], High: [] },
                   Furniture: { Low: [], Medium: [], High: [] }
               };
           }
           
           Object.entries(location.componentResults).forEach(([component, specs]) => {
               Object.entries(specs).forEach(([spec, rate]) => {
                   if (typeof rate === 'number' && rate > 0) {
                       allRates[component][spec].push(rate);
                       consistencyAnalysis.averagesByRegion[region][component][spec].push(rate);
                   } else if (component === 'Technology' && typeof rate === 'object') {
                       // Handle layout-specific Technology rates
                       Object.values(rate).forEach(layoutRate => {
                           if (typeof layoutRate === 'number' && layoutRate > 0) {
                               allRates[component][spec].push(layoutRate);
                               consistencyAnalysis.averagesByRegion[region][component][spec].push(layoutRate);
                           }
                       });
                   }
               });
           });
       });
       
       // Calculate global statistics
       Object.entries(allRates).forEach(([component, specs]) => {
           consistencyAnalysis.rateRanges[component] = {};
           Object.entries(specs).forEach(([spec, rates]) => {
               if (rates.length > 0) {
                   const sortedRates = rates.sort((a, b) => a - b);
                   consistencyAnalysis.rateRanges[component][spec] = {
                       min: sortedRates[0],
                       max: sortedRates[sortedRates.length - 1],
                       average: rates.reduce((sum, rate) => sum + rate, 0) / rates.length,
                       median: sortedRates[Math.floor(sortedRates.length / 2)],
                       count: rates.length
                   };
               }
           });
       });
       
       // Calculate regional averages
       Object.entries(consistencyAnalysis.averagesByRegion).forEach(([region, components]) => {
           Object.entries(components).forEach(([component, specs]) => {
               Object.entries(specs).forEach(([spec, rates]) => {
                   if (rates.length > 0) {
                       consistencyAnalysis.averagesByRegion[region][component][spec] = {
                           average: rates.reduce((sum, rate) => sum + rate, 0) / rates.length,
                           count: rates.length
                       };
                   } else {
                       consistencyAnalysis.averagesByRegion[region][component][spec] = null;
                   }
               });
           });
       });
       
       console.log("Rate Ranges by Component:", consistencyAnalysis.rateRanges);
       console.log("Regional Averages:", consistencyAnalysis.averagesByRegion);
       
       return consistencyAnalysis;
   }

8. Identify and flag rate outliers:

   function identifyRateOutliers(extractionResults, consistencyAnalysis) {
       console.log("=== IDENTIFYING RATE OUTLIERS ===");
       
       const outliers = [];
       
       extractionResults.successful.forEach(location => {
           Object.entries(location.componentResults).forEach(([component, specs]) => {
               Object.entries(specs).forEach(([spec, rate]) => {
                   if (typeof rate === 'number' && rate > 0) {
                       const globalStats = consistencyAnalysis.rateRanges[component][spec];
                       
                       if (globalStats) {
                           const average = globalStats.average;
                           const standardDeviation = Math.sqrt(
                               extractionResults.successful
                                   .flatMap(loc => {
                                       const locRate = loc.componentResults[component]?.[spec];
                                       return typeof locRate === 'number' ? [locRate] : [];
                                   })
                                   .reduce((sum, r) => sum + Math.pow(r - average, 2), 0) /
                               globalStats.count
                           );
                           
                           // Flag as outlier if more than 2 standard deviations from mean
                           if (Math.abs(rate - average) > 2 * standardDeviation) {
                               outliers.push({
                                   location: `${location.country}-${location.city}`,
                                   component: component,
                                   specification: spec,
                                   rate: rate,
                                   average: average,
                                   deviation: Math.abs(rate - average) / standardDeviation
                               });
                           }
                       }
                   }
               });
           });
       });
       
       console.log(`Found ${outliers.length} rate outliers:`);
       outliers.forEach(outlier => {
           console.log(`  ${outlier.location}: ${outlier.component} ${outlier.specification} = $${outlier.rate} (${outlier.deviation.toFixed(2)} std dev from mean)`);
       });
       
       return outliers;
   }
