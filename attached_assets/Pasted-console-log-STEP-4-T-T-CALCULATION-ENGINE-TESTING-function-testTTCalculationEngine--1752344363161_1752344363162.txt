console.log("=== STEP 4: T&T CALCULATION ENGINE TESTING ===");

function testTTCalculationEngine() {
    console.log("Testing T&T calculation engine with working data...");
    
    // Use the working location and rates from Step 3
    const testLocation = step3Results.testLocation;
    const workingRates = step3Results.rateTests;
    
    // Find a working component/spec combination
    let workingComponent = null;
    let workingSpec = null;
    
    for (const [component, specs] of Object.entries(workingRates)) {
        if (component === "Professional Fees") continue;
        
        for (const [spec, result] of Object.entries(specs)) {
            if (result.success) {
                workingComponent = component;
                workingSpec = spec;
                break;
            }
        }
        if (workingComponent) break;
    }
    
    if (!workingComponent) {
        console.error("❌ No working component/spec combination found");
        return { error: "No working rates available for calculation test" };
    }
    
    console.log(`Testing calculation with: ${testLocation}, ${workingComponent}, ${workingSpec}`);
    
    try {
        // Test simple calculation
        const testInputs = {
            location: testLocation,
            area: 1000,  // Simple area for testing
            component: workingComponent,
            specification: workingSpec
        };
        
        const calculationResult = calculateTTComponentCost(testInputs);
        console.log(`Calculation result: $${calculationResult}`);
        
        // Verify calculation logic
        const expectedRate = workingRates[workingComponent][workingSpec].rate;
        const expectedCost = expectedRate * testInputs.area;
        const calculationCorrect = Math.abs(calculationResult - expectedCost) < 1;
        
        console.log(`Expected: $${expectedCost}, Got: $${calculationResult}, Correct: ${calculationCorrect}`);
        
        return {
            testInputs: testInputs,
            calculationResult: calculationResult,
            expectedCost: expectedCost,
            calculationCorrect: calculationCorrect,
            success: !!(calculationResult && calculationResult > 0 && calculationCorrect)
        };
        
    } catch (error) {
        console.error("❌ T&T calculation failed:", error);
        return { error: error.message };
    }
}

// Execute Step 4
const step4Results = testTTCalculationEngine();
console.log("STEP 4 RESULTS:", step4Results);

if (!step4Results.success) {
    console.error("❌ STEP 4 FAILED - T&T calculation engine not working");
    throw new Error("T&T calculation engine failed");
}

console.log("✅ STEP 4 PASSED - T&T calculation engine working");
