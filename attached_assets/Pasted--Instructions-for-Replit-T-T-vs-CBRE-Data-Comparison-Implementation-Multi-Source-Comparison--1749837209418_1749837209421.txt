## 3. Location Intersection Logic

### **Smart Location Matching:**

```javascript
function getComparableLocations() {
  const ttLocations = getTTAvailableLocations();
  const cbreLocations = getCBREAvailableLocations();
  
  // Find cities available in both datasets
  const commonCities = ttLocations.filter(ttLoc => 
    cbreLocations.some(cbreLoc => 
      ttLoc.city === cbreLoc.city && ttLoc.country === cbreLoc.country
    )
  );
  
  // If no common cities, use regional averages
  if (commonCities.length === 0) {
    return getRegionalAverages(ttLocations, cbreLocations);
  }
  
  return commonCities;
}
```

## 5. Comparative Calculation Engine

### **Dual Calculation Process:**

```javascript
function generateComparativeBudget(projectData) {
  const mappedComponents = mapSelectedComponents(projectData.scope);
  
  const ttBudget = calculateTTBudget({
    location: projectData.location,
    specification: projectData.specification,
    components: mappedComponents.tt,
    area: projectData.area
  });
  
  const cbreBudget = calculateCBREBudget({
    location: projectData.location,
    specification: projectData.specification,
    components: mappedComponents.cbre,
    area: projectData.area
  });
  
  return {
    tt: ttBudget,
    cbre: cbreBudget,
    comparison: generateVarianceAnalysis(ttBudget, cbreBudget),
    methodology: getMethodologyDifferences()
  };
}
```

## 7. Variance Analysis & Insights

### **Automated Analysis:**

```javascript
function generateInsights(ttBudget, cbreBudget) {
  const totalVariance = Math.abs(ttBudget.total - cbreBudget.total);
  const percentageVariance = (totalVariance / Math.min(ttBudget.total, cbreBudget.total)) * 100;
  
  return {
    summary: `${percentageVariance.toFixed(1)}% variance between sources`,
    highestVariance: findHighestVarianceComponent(ttBudget, cbreBudget),
    recommendation: generateRecommendation(percentageVariance),
    reliability: assessDataReliability(ttBudget.location, cbreBudget.location)
  };
}
```
