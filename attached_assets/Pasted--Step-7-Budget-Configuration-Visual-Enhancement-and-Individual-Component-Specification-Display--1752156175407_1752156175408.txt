### Phase 3: Specification Level Mapping
```
5. Create proper specification level mapping:

   function getSpecificationDisplay(specLevel) {
       const specMap = {
           'Low': 'Bronze (Low)',
           'Medium': 'Silver (Medium)', 
           'High': 'Gold (High)',
           'Bronze': 'Bronze (Low)',
           'Silver': 'Silver (Medium)',
           'Gold': 'Gold (High)'
       };
       return specMap[specLevel] || specLevel;
   }

6. Handle Professional Fees specification display:
   
   function formatComponentSpecification(componentName, specification) {
       if (componentName === 'Professional Fees') {
           return 'Calculated as % of other components';
       }
       return getSpecificationDisplay(specification);
   }

7. Calculate and display individual rates correctly:
   
   function formatComponentRate(componentName, rate, percentage = null) {
       if (componentName === 'Professional Fees') {
           return `${percentage}% of other components`;
       }
       return `US$${rate.toLocaleString()}/sqm`;
   }
```
